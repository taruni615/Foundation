pymysql           # MySQL driver for the qa_* tables (no ORM)
pypdf             # PDF page extraction for the chapter/topic export endpoints
xhtml2pdf         # HTML -> PDF chapter export (viewer_api)
fpdf2             # Fallback PDF writer used when xhtml2pdf cannot render

pandas
openpyxl
